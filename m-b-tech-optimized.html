{

    METHOD_LIST_BEGIN
    METHOD_ADD(ApiController, get, "/api/v1/data", Get);
    METHOD_ADD(ApiController, post, "/api/v1/process", Post);
    METHOD_LIST_END

    void get(const HttpRequestPtr& req,
             std::function<void(const HttpResponsePtr&)> callback);
